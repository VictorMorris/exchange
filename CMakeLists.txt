cmake_minimum_required(VERSION 3.20)
project(exchange CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(-Wall -Wextra -Wconversion -Wsign-conversion)

add_library(engine src/book.cpp src/engine.cpp)
target_include_directories(engine PUBLIC src)

enable_testing()
add_executable(book_test tests/book_test.cpp)
target_link_libraries(book_test PRIVATE engine)
target_include_directories(book_test PRIVATE tests)
add_test(NAME book COMMAND book_test)
add_executable(engine_test tests/engine_test.cpp)
target_link_libraries(engine_test PRIVATE engine)
target_include_directories(engine_test PRIVATE tests)
add_test(NAME engine COMMAND engine_test)
